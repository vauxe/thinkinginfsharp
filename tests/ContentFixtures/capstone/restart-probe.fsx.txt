#r "Booking.Domain.dll"
#r "Booking.Contracts.dll"
#r "Booking.Infrastructure.dll"

open System
open System.Threading
open System.Threading.Tasks
open Booking.Domain
open Booking.Domain.Ports
open Booking.Infrastructure

let fail message =
    eprintfn "%s" message
    exit 1

let expectOk result =
    match result with
    | Ok value -> value
    | Error error -> failwithf "Expected Ok, received Error %A" error

match fsi.CommandLineArgs |> Array.skip 1 with
| [| snapshotPath; rawEventId; rawCapacity; rawRequestId; rawSeats |] ->
    let eventId = EventId.create rawEventId |> expectOk
    let capacity = Capacity.create (Int32.Parse rawCapacity) |> expectOk
    let event = Event.create eventId capacity
    let configuration = BookingStoreConfiguration.create snapshotPath |> expectOk
    let store = AtomicBookingStore configuration

    let forbiddenPayment (_: PaymentRequest) (_: CancellationToken) =
        Task.FromException<PaymentOutcome>(InvalidOperationException "payment must not be replayed")

    let forbiddenNotification (_: NotificationRequest) (_: CancellationToken) =
        Task.FromException<unit>(InvalidOperationException "notification must not be replayed")

    let application =
        IdempotentBookingService(event, store, forbiddenPayment, forbiddenNotification)

    match
        application
            .Execute(BookingCommand.Place(Commands.place rawRequestId (Int32.Parse rawSeats)), CancellationToken.None)
            .GetAwaiter()
            .GetResult()
    with
    | Error error -> fail $"Replay failed: {error}"
    | Ok booking ->
        let status =
            match Booking.status booking with
            | BookingStatus.Pending -> "pending"
            | BookingStatus.Confirmed _ -> "confirmed"
            | BookingStatus.Cancelled _ -> "cancelled"

        printfn
            "restored|%s|%d|%s"
            (booking |> Booking.requestId |> RequestId.value)
            (booking |> Booking.seats |> SeatCount.value |> int)
            status
| _ -> fail "Usage: restart-probe.fsx SNAPSHOT EVENT_ID CAPACITY REQUEST_ID SEATS"
